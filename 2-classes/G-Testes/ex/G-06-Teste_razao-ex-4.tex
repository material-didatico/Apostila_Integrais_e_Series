\begin{question}
  Use o teste da razão para verificar a convergência ou divergência da série
  \[
    a_1 = 5,
    \quad
    a_{n+1} = \frac{\,\sqrt[n]{n}\,}{2} a_n
  \]
\end{question}

%------------------------------------------------------------------------------%
\begin{resolution}{Solução}
  \[
    a_{n+1} = \frac{\,\sqrt[n]{n}\,}{2} a_n
  \]

  \pause
  \[
    \frac{a_{n+1}}{a_n}
    = \frac{\sqrt[n]{n}}{2}
  \]

  \pause
  \[
    \rho
    \pause
    \;=\; \lim \frac{a_{n+1}}{a_n}
    \pause
    \;=\; \lim \frac{\sqrt[n]{n}}{2}
    \pause
    \;=\; \frac{1}{2}
    \pause
    \;<\; 1
  \]

  \pause
  A série converge pelo teste da razão
\end{resolution}
