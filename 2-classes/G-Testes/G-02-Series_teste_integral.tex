\section{Série com Termos Não-Negativos}

%------------------------------------------------------------------------------%
\begin{frame}{Recapitulando}
  Sequências numéricas
  \[
    \big( a_k \big)_{k=1}^\infty \qquad\qquad\qquad a_1,\; a_2,\; a_3,\; \dots
  \]
  \pause
  Séries numéricas
  \[
    \sum_{k=1}^\infty a_k = a_1 + a_2 + a_3 + \cdots + a_k + \cdots
  \]
  \pause
  Somas parciais
  \[
    S_n = \sum_{k=1}^n a_k = a_1 + a_2 + a_3 + \cdots + a_n
  \]
\end{frame}

%------------------------------------------------------------------------------%
\begin{frame}{Série com Termos Não-Negativos}
  Caso particular de séries onde $\quad a_n \geq 0 \quad \forall n$

  \pause
  \bigskip
  Uma série
  \[\sum_{n=1}^\infty a_n \qquad \text{com} \qquad a_n\geq 0 \]

  \pause
  \bigskip
  é \emph{convergente} se, e somente se, sua sequência de somas parciais,  \\
  $S_n$, é uma sequência \emph{limitada superiormente}.
\end{frame}

\section{Teste da Integral}

%------------------------------------------------------------------------------%
\begin{frame}{Teste da Integral}
  Trocamos o problema de verificar se uma série converge para o de
  verificar se uma integral converge
  \[
    \sum_{n=1}^{\infty} a_n  \;\text{ converge?}
    \qquad \text{ por } \qquad
    \int_{1}^{\infty} f(x) \,dx  \;\text{ converge?}
  \]

  \pause
  O valor da integral \emph{\textbf{não é}} uma boa aproximação para a soma da série
  \[ \sum_{n=1}^{\infty} a_n  \quad \not\approx \quad \int_{1}^{\infty} f(x) \,dx\]
\end{frame}

%------------------------------------------------------------------------------%
\begin{frame}[t]{Teste da Integral}
  \onslide<+->{Seja $(a_n)$ uma sequência tal que}
  \begin{enumerate}[<+->]
    \item seus termos são positivos, $a_n>0$
    \item $a_n=f(n)$
    \item \emph{a função $f$} é contínua, positiva e decrescente para todo $x \geq N$
  \end{enumerate}

	\bigskip
  \onslide<+->{Então, a série \tabto{26mm}\(\sum_{n=N}^\infty a_n\) \tabto{50mm} converge}

  \smallskip
	\onslide<+->{\emph{se, e somente se}}

  \medskip
  \onslide<+->{a integral     \tabto{26mm}\(\int_N^\infty f(x)dx \) \tabto{50mm} converge}
\end{frame}

%------------------------------------------------------------------------------%
\begin{frame}{Ilustração do Teste da Integral -- 1}
  \begin{center}
    \scalebox{.9}{\input{fig_teste_integral_right}}
  \end{center}

  \pause
  A série é \emph{``maior''} do que a integral
  \pause
  \(
    \qquad\qquad
    \int_{N}^{\infty} f(x)\,dx
    \quad\leq\quad
    \sum_{n=N}^{\infty} a_n
  \)
\end{frame}

%------------------------------------------------------------------------------%
\begin{frame}{Ilustração do Teste da Integral -- 2}
  \begin{center}
    \scalebox{.9}{\input{fig_teste_integral_left}}
  \end{center}

  \pause
  A série é \emph{``menor''} do que a integral
  \pause
  \(
    \qquad\qquad
    \sum_{n=N}^\infty a_k
    \quad\leq\quad
    a_{N} + \int_{N}^\infty f(x)\,dx
  \)
\end{frame}

%------------------------------------------------------------------------------%
\begin{frame}{Comparação entre a Série e a Integral}
  Temos portanto as relações
  \[
    \int_{N}^{\infty} f(x)\,dx
    \quad\leq\quad
    \sum_{k=N}^{\infty} a_k
    \quad\leq\quad
    a_N + \int_{N}^{\infty} f(x)\,dx
  \]
\end{frame}

%------------------------------------------------------------------------------%
\begin{frame}{Lembrando como Calcular uma Integral Imprópria}
  \[
    \int_{a}^{\infty} f(x)\,dx
    \pause \;=\;
    \lim_{b\to\infty} \; \int_{a}^{b} f(x)\,dx
    \pause \;=\;
    \lim_{b\to\infty} \left( F(x) \evalat{a}{b} \; \right)
    \pause \;=\;
    \lim_{b\to\infty} \big( F(b) - F(a) \big)
  \]
\end{frame}

\section{Estimativa de Erro}

%------------------------------------------------------------------------------%
\begin{frame}{Estimativa de Erro}
  Não temos uma fórmula para a soma da série

  \pause
  \vspace{\baselineskip}
  Podemos usar as somas parciais para aproximar a soma

  \pause
  \vspace{\baselineskip}
  Quantos termos precisamos?
\end{frame}

%------------------------------------------------------------------------------%
\begin{frame}{Resto ou Erro}
  \begin{align*}
    \action<+->{S & = \sum_{k=1}^{\infty} a_k} \\[2mm]
    \action<+->{  & = \sum_{k=1}^{n} a_k + \sum_{k=n+1}^{\infty} a_k} \\[2mm]
    \action<+->{  & = S_n + R_n}
  \end{align*}
  \action<+->{
  $R_n$ é o \emph{resto da série}, isso é, o que faltou ser calculado
  \[ R_n = \sum_{k=n+1}^{\infty} a_k \]}
\end{frame}

%------------------------------------------------------------------------------%
\begin{frame}{Estimativa de Erro}
  Se uma série é \emph{convergente pelo teste da integral} podemos estimar o resto
  \[ R_n = a_{n+1} + a_{n+2} + a_{n+3} + a_{n+4} + \cdots \]
  \pause
  comparando a área dos retângulos com a da integral temos
  \pause
  \[ R_n \geq \int_{n+1}^{\infty} f(x)\,dx \]
  \pause
  \[ R_n \leq \int_{n}^{\infty} f(x)\,dx \]
\end{frame}

%------------------------------------------------------------------------------%
\begin{frame}{Limitantes para o resto no teste da integral}
  Se a série
  \(
    \quad \sum_{k=1}^{\infty} a_k
  \)

  \vspace{0.5\baselineskip}
  converge segundo o teste da integral, então o resto satisfaz
  \[
    \int_{n+1}^{\infty} f(x)\,dx
    \quad\leq\quad
    R_n
    \quad\leq\quad
    \int_{n}^{\infty} f(x)\,dx
  \]
\end{frame}
