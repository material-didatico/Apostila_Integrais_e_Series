%------------------------------------------------------------------------------%
\begin{question}
  Sendo $0<r<1$, calcule o limite da sequência
  \(
    a_k = \dfrac{\;-\dfrac{1}{r^k}+7\;}{ 1 + \dfrac{5}{r^k}}
  \)
\end{question}

%------------------------------------------------------------------------------%
\begin{resolution}{Solução}
  Como $0<r<1$, sabemos que, \(\lim r^k=0\)

  \[
    \pause
    a_k
    \;=\; \frac{-\dfrac{1}{r^k}+7}{1+ \dfrac{5}{r^k}}
    \pause
    \;=\; \frac{\dfrac{-1+7r^k}{r^k}}{\dfrac{r^k + 5}{r^k}}
    \pause
    \;=\; \frac{-1+7r^k}{r^k+5}
  \]
\end{resolution}

%------------------------------------------------------------------------------%

%------------------------------------------------------------------------------%
\begin{resolution}{Solução}
  \[
    \lim a_k
    \pause
    \;=\; \lim \left(\frac{-1+7r^k}{r^k+5}\right)
    \pause
    \;=\; \dfrac{-1+7\lim(r^k)}{\lim(r^k) + 5}
    \pause
    \;=\; \frac{-1+7\times 0}{0+5}
    \pause
    \;=\;-\frac{1}{5}
  \]
\end{resolution}
