\begin{question}
Dada a fórmula para o $n$-ésimo termo, $a_n$, de uma sequência $(a_n)$,\\
encontre os valores de $a_1$, $a_2$, $a_3$ e $a_4$.
\[
  a_n = \frac{1-n}{n^2}
\]
\end{question}

%------------------------------------------------------------------------------%
\begin{resolution}{Solução}
\begin{align*}
  \onslide<+->{a_1 }
  \onslide<+->{& = \dfrac{\;1-n\;}{n^2}\evalat{n=1}{} }
  \onslide<+->{= \dfrac{\;1-1\;}{1^2} }
  \onslide<+->{= 0              \\[3mm]}
  \onslide<+->{a_2 }
  \onslide<+->{& = \dfrac{\;1-n\;}{n^2}\evalat{n=2}{} }
  \onslide<+->{= \dfrac{\;1-2\;}{2^2} }
  \onslide<+->{= -\dfrac{1}{4}  \\[3mm]}
  \onslide<+->{a_3 }
  \onslide<+->{& = \dfrac{\;1-n\;}{n^2}\evalat{n=3}{} }
  \onslide<+->{= \dfrac{\;1-3\;}{3^2} }
  \onslide<+->{= -\dfrac{2}{9}  \\[3mm]}
  \onslide<+->{a_4 }
  \onslide<+->{& = \dfrac{\;1-n\;}{n^2}\evalat{n=4}{} }
  \onslide<+->{= \dfrac{\;1-4\;}{4^2} }
  \onslide<+->{= -\dfrac{3}{16}}
\end{align*}
\end{resolution}
