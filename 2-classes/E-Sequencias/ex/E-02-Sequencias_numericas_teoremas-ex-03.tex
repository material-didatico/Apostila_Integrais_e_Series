%------------------------------------------------------------------------------%
\begin{question}
  Mostre que \( \lim\frac{5k^4}{k^4+8k^3} = 5 \)
\end{question}

%------------------------------------------------------------------------------%
\begin{resolution}{Solução}
  A função
  \[
    f(x) = \frac{5x^4}{x^4+8x^3}
    \hspace{50mm}
    \emph{x\in\R}
  \]
  está definida para todo $x>0$
  \pause
  e a sequência
  \[
    a_k = f(k)=\frac{5k^4}{k^4+8k^3}
    \hspace{40.5mm}
    \emph{k\in\N}
  \]
\end{resolution}

%------------------------------------------------------------------------------%
\begin{resolution}{Solução}
  \onslide<+->{Podemos aplicar a regra de L'Hôpital \emph{em $x$},}
  \onslide<+->{pois temos uma indeterminação $\sfrac{\infty}{\infty}$}
  \begin{align*}
    \onslide<+->{    \lim_{k\to\infty} \frac{5k^4}{k^4+8k^3}}
    \onslide<+->{  = \emph{\lim_{x\to\infty} \frac{5x^4}{x^4+8x^3}}}
    \onslide<+->{& = \lim_{x\to\infty} \frac{20x^3}{4x^3+24x^2} \\}
    \onslide<+->{& = \lim_{x\to\infty} \frac{60x^2}{12x^2+48x}  \\}
    \onslide<+->{& = \lim_{x\to\infty} \frac{120x}{24x+48}      \\}
    \onslide<+->{& = \lim_{x\to\infty} \frac{120}{24}           \\}
    \onslide<+->{& = 5}
  \end{align*}
\end{resolution}

%------------------------------------------------------------------------------%
\begin{resolution}{Método mais apropriado}
  \begin{align*}
    \onslide<+->{    \lim_{k\to\infty} \frac{5k^4}{k^4+8k^3}}
    \onslide<+->{& = \lim_{k\to\infty} \frac{\dfrac{\,5k^4\,}{k^4}}{\;\dfrac{\;k^4+8k^3\;}{k^4}\;} \\}
    \onslide<+->{& = \lim_{k\to\infty} \frac{5}{1+\dfrac{8}{k}} \\}
    \onslide<+->{& = \frac{5}{\displaystyle 1 + 8\lim_{k\to\infty} \frac{1}{k}} \\}
    \onslide<+->{& = 5}
  \end{align*}
\end{resolution}
