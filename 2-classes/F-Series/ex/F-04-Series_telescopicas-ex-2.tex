\begin{question}

 Calcular a soma da série
	\[
    \sum_{n=1}^{\infty} \frac{1}{\;n(n+1)\;}
  \]

\end{question}

%------------------------------------------------------------------------------%
\begin{resolution}{Pode Não Ser Óbvio}

  Usando frações parciais
  \[
    \frac{1}{n(n+1)} = \frac{1}{n} - \frac{1}{n+1}
  \]

  \pause
  Temos
  \[
    \sum_{n=1}^{\infty} \frac{1}{\;n(n+1)\;}
    \;=\; \sum_{n=1}^{\infty} \left( \frac{1}{n} - \frac{1}{n+1} \right)
    \;=\; 1
  \]
\end{resolution}
